# MODÈLE LOGIQUE DE DONNÉES (MLD) - BRASIL BURGER

## TABLES

1. client

client (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom VARCHAR(100) NOT NULL,
  prenom VARCHAR(100) NOT NULL,
  telephone VARCHAR(20) UNIQUE NOT NULL,
  email VARCHAR(150) UNIQUE NOT NULL,
  password VARCHAR(255) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)


2. gestionnaire

gestionnaire (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom VARCHAR(100) NOT NULL,
  prenom VARCHAR(100) NOT NULL,
  email VARCHAR(150) UNIQUE NOT NULL,
  password VARCHAR(255) NOT NULL,
  role VARCHAR(50) DEFAULT 'GESTIONNAIRE',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)


3. livreur

livreur (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom VARCHAR(100) NOT NULL,
  prenom VARCHAR(100) NOT NULL,
  telephone VARCHAR(20) NOT NULL,
  email VARCHAR(150) UNIQUE NOT NULL,
  password VARCHAR(255) NOT NULL,
  disponible BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)



burger (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom VARCHAR(100) NOT NULL,
  description TEXT,
  prix DECIMAL(10, 2) NOT NULL,
  image_url VARCHAR(255),
  disponible BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)


5. complement

complement (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom VARCHAR(100) NOT NULL,
  type VARCHAR(20) NOT NULL CHECK (type IN ('BOISSON', 'FRITE')),
  prix DECIMAL(10, 2) NOT NULL,
  disponible BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)


6. menu

menu (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom VARCHAR(100) NOT NULL,
  description TEXT,
  burger_id UUID NOT NULL REFERENCES burger(id) ON DELETE CASCADE,
  boisson_id UUID NOT NULL REFERENCES complement(id) ON DELETE RESTRICT,
  frite_id UUID NOT NULL REFERENCES complement(id) ON DELETE RESTRICT,
  prix_total DECIMAL(10, 2) NOT NULL,
  disponible BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)


7. zone

zone (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom VARCHAR(100) NOT NULL UNIQUE,
  prix_livraison DECIMAL(10, 2) NOT NULL,
  description TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)


8. quartier

quartier (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom VARCHAR(100) NOT NULL,
  zone_id UUID NOT NULL REFERENCES zone(id) ON DELETE CASCADE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(nom, zone_id)
)

9. commande

commande (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  numero_commande VARCHAR(50) UNIQUE NOT NULL,
  client_id UUID NOT NULL REFERENCES client(id) ON DELETE CASCADE,
  date_commande TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  montant_total DECIMAL(10, 2) NOT NULL,
  etat VARCHAR(20) NOT NULL DEFAULT 'EN_ATTENTE'
    CHECK (etat IN ('EN_ATTENTE', 'VALIDEE', 'EN_PREPARATION', 'TERMINEE', 'EN_LIVRAISON', 'LIVREE', 'ANNULEE')),
  zone_id UUID REFERENCES zone(id) ON DELETE SET NULL,
  livreur_id UUID REFERENCES livreur(id) ON DELETE SET NULL,
  adresse_livraison TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)

10. ligne_commande

ligne_commande (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  commande_id UUID NOT NULL REFERENCES commande(id) ON DELETE CASCADE,
  type_item VARCHAR(20) NOT NULL CHECK (type_item IN ('BURGER', 'MENU')),
  burger_id UUID REFERENCES burger(id) ON DELETE RESTRICT,
  menu_id UUID REFERENCES menu(id) ON DELETE RESTRICT,
  quantite INTEGER NOT NULL CHECK (quantite > 0),
  prix_unitaire DECIMAL(10, 2) NOT NULL,
  sous_total DECIMAL(10, 2) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CHECK (
    (type_item = 'BURGER' AND burger_id IS NOT NULL AND menu_id IS NULL) OR
    (type_item = 'MENU' AND menu_id IS NOT NULL AND burger_id IS NULL)
  )
)

11. paiement

paiement (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  commande_id UUID UNIQUE NOT NULL REFERENCES commande(id) ON DELETE CASCADE,
  date_paiement TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  montant DECIMAL(10, 2) NOT NULL,
  mode_paiement VARCHAR(10) NOT NULL CHECK (mode_paiement IN ('WAVE', 'OM')),
  reference_transaction VARCHAR(100) UNIQUE NOT NULL,
  statut VARCHAR(20) DEFAULT 'EN_ATTENTE' CHECK (statut IN ('EN_ATTENTE', 'REUSSI', 'ECHOUE')),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)

CONTRAINTES D'INTÉGRITÉ

Clés Primaires (PK)
- Toutes les tables ont un id UUID comme clé primaire

Clés Étrangères (FK)
- menu.burger_id → burger.id
- menu.boisson_id → complement.id
- menu.frite_id → complement.id
- quartier.zone_id → zone.id
- commande.client_id → client.id
- commande.zone_id → zone.id (nullable)
- commande.livreur_id → livreur.id (nullable)
- ligne_commande.commande_id → commande.id
- ligne_commande.burger_id → burger.id (nullable)
- ligne_commande.menu_id → menu.id (nullable)
- paiement.commande_id → commande.id (unique)

Contraintes Uniques
- client.telephone
- client.email
- gestionnaire.email
- livreur.email
- zone.nom
- commande.numero_commande
- paiement.commande_id
- paiement.reference_transaction
- quartier(nom, zone_id) - composite

Contraintes de Validation (CHECK)
- complement.type IN ('BOISSON', 'FRITE')
- commande.etat IN ('EN_ATTENTE', 'VALIDEE', 'EN_PREPARATION', 'TERMINEE', 'EN_LIVRAISON', 'LIVREE', 'ANNULEE')
- ligne_commande.type_item IN ('BURGER', 'MENU')
- ligne_commande.quantite > 0
- ligne_commande: Vérification de cohérence burger_id/menu_id selon type_item
- paiement.mode_paiement IN ('WAVE', 'OM')
- paiement.statut IN ('EN_ATTENTE', 'REUSSI', 'ECHOUE')


 CARDINALITÉS

- Client (1,n) -- passe -- (0,n) Commande
- Commande (1,1) -- contient -- (1,n) LigneCommande
- Burger (1,1) -- est référencé -- (0,n) LigneCommande
- Menu (1,1) -- est référencé -- (0,n) LigneCommande
- Burger (1,1) -- compose -- (0,n) Menu
- Complement (1,1) -- compose -- (0,n) Menu
- Commande (0,1) -- a -- (1,1) Paiement
- Zone (1,1) -- couvre -- (1,n) Quartier
- Zone (1,1) -- contient -- (0,n) Commande
- Livreur (1,1) -- livre -- (0,n) Commande
